cmake_minimum_required(VERSION 3.22.1)
project("didawmidinative")

set(CMAKE_CXX_STANDARD 17)

include(FetchContent)

    oboe
    GIT_REPOSITORY https://github.com
    GIT_TAG tags/v1.8.0
)

add_library(didawmidinative SHARED
    midi_daw_engine.cpp
)

target_link_libraries(didawmidinative
    oboe
    android
    amidi
    log
)
